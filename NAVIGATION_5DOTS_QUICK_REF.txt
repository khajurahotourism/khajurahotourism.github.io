  BEFORE: onClick={() => carouselApi?.scrollTo(index)}
  AFTER:  onClick={() => {

              Math.floor(testimonials.length / 5);
            carouselApi?.scrollTo(targetReview);
          }}

  BEFORE: index === activeSlide
  AFTER:  dotIndex === Math.floor(
            activeSlide / Math.floor(testimonials.length / 5))
